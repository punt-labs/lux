% FIDELITY CONTROL for docs/menu_presence.tex, defect D5.
%
% Differs from menu_presence.tex in exactly one place: Identify is split into
% IdentifyRead and IdentifyWrite around the store write lock. The shipped
% facade reads `before = identity.current(scope)` outside the lock, then
% identify_client, then MenuArming.track(conn, before), which binds the sink
% and marks only if the identity changed from `before`. A departure that lands
% between the read and the write makes the re-registration look unchanged.
% (iphase/ifor/ibefore are the split-window devices, inert in the fixed spec.)
%
% Expected: BOTH negations are REACHABLE.
%   P1: probcli docs/menu_presence_d5_buggy.tex -model_check -nodead \
%         -goal "dirty = fclear & (not((earns \ lapsed) <: shown) or not(shown <: earns))" -p DEFAULT_SETSIZE 3
%   P2: probcli docs/menu_presence_d5_buggy.tex -model_check -nodead \
%         -goal "#c.(c : CONN & ((c : earns & sinks(c) /= 1) or sinks(c) > 1 \
%           or (c /: registered & sinks(c) /= 0)))" -p DEFAULT_SETSIZE 3
% Trace: Attach(c); IdentifyRead(c) [before = earning]; DropGraceful(c); Push;
% IdentifyWrite(c) leaves earns = {c}, sinks(c) = 0, dirty = fclear,
% shown = {}: no entry now, and a ghost when c next departs.
%
% The remaining operations are those of menu_presence.tex, whose prose
% describes them.
%

\documentclass[a4paper,10pt,fleqn]{article}
\usepackage[margin=1in]{geometry}
\usepackage{fuzz}

\begin{document}

\title{lux Menu Presence, D5 control: identify's before-snapshot outside the lock}
\maketitle

\begin{zed}
  [CONN]
\end{zed}

\begin{zed}
  FSTATE ::= fclear | fset
\end{zed}

\begin{zed}
  IPHASE ::= iidle | iwrite
\end{zed}

\begin{schema}{MenuPres}
  registered : \power CONN \\
  earns : \power CONN \\
  lapsed : \power CONN \\
  open : \power CONN \\
  sinks : CONN \fun 0 \upto 2 \\
  dirty : FSTATE \\
  shown : \power CONN \\
  iphase : IPHASE \\
  ifor : \power CONN \\
  ibefore : FSTATE
\where
  earns \subseteq registered \\
  lapsed \subseteq registered \\
  \# ifor \leq 1
\end{schema}

\begin{schema}{Init}
  MenuPres'
\where
  registered' = \emptyset \\
  earns' = \emptyset \\
  lapsed' = \emptyset \\
  open' = \emptyset \\
  sinks' = CONN \cross \{ 0 \} \\
  dirty' = fclear \\
  shown' = \emptyset \\
  iphase' = iidle \\
  ifor' = \emptyset \\
  ibefore' = fclear
\end{schema}

\begin{schema}{AgentRecord}
  \Delta MenuPres \\
  c? : CONN
\where
  registered' = registered \cup \{ c? \} \\
  earns' = earns \cup \{ c? \} \\
  lapsed' = lapsed \setminus \{ c? \} \\
  sinks' = sinks \oplus \{ c? \mapsto 1 \} \\
  (c? \in earns \land c? \notin lapsed \land dirty' = dirty) \lor
    ((c? \notin earns \lor c? \in lapsed) \land dirty' = fset) \\
  shown' = shown \\
  iphase' = iphase \\
  ifor' = ifor \\
  ibefore' = ibefore
\end{schema}

\begin{schema}{IdentifyRead}
  \Delta MenuPres \\
  c? : CONN
\where
  iphase = iidle \\
  iphase' = iwrite \\
  ifor' = \{ c? \} \\
  (c? \in earns \land ibefore' = fset) \lor
    (c? \notin earns \land ibefore' = fclear) \\
  registered' = registered \\
  earns' = earns \\
  lapsed' = lapsed \\
  open' = open \\
  sinks' = sinks \\
  dirty' = dirty \\
  shown' = shown
\end{schema}

\texttt{IdentifyRead} is \texttt{Operations.identify}'s
\texttt{before = identity.current(scope)}, taken outside the store's write
lock: it snapshots whether the connection already earns presence.

\begin{schema}{IdentifyWrite}
  \Delta MenuPres \\
  c? : CONN
\where
  iphase = iwrite \\
  c? \in ifor \\
  iphase' = iidle \\
  ifor' = \emptyset \\
  ibefore' = fclear \\
  registered' = registered \cup \{ c? \} \\
  earns' = earns \cup \{ c? \} \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \\
  (ibefore = fclear \land sinks' = sinks \oplus \{ c? \mapsto 1 \}) \lor
    (ibefore = fset \land sinks' = sinks) \\
  (ibefore = fclear \land dirty' = fset) \lor
    (ibefore = fset \land dirty' = dirty) \\
  shown' = shown
\end{schema}

\texttt{IdentifyWrite} is \texttt{identify\_client} followed by
\texttt{MenuArming.track(conn, before)}, which binds the sink and marks only
if the identity differs from \texttt{before}. A departure landing between the
two makes the re-registration look unchanged: the defect.

\begin{zed}
  Attach \defs AgentRecord \land [\Delta MenuPres; c? : CONN | open' = open \cup \{ c? \}]
\end{zed}

\begin{zed}
  ReadLoopRecord \defs AgentRecord \land
    [\Delta MenuPres; c? : CONN | open' = open \land c? \in open]
\end{zed}



\begin{schema}{Connect}
  \Delta MenuPres \\
  c? : CONN
\where
  registered' = registered \cup \{ c? \} \\
  earns' = earns \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \\
  (c? \in earns \land sinks' = sinks \oplus \{ c? \mapsto 1 \}) \lor
    (c? \notin earns \land sinks' = sinks) \\
  (c? \in earns \land c? \in lapsed \land dirty' = fset) \lor
    (\lnot (c? \in earns \land c? \in lapsed) \land dirty' = dirty) \\
  shown' = shown \\
  iphase' = iphase \\
  ifor' = ifor \\
  ibefore' = ibefore
\end{schema}

\begin{schema}{Renew}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \in registered \\
  registered' = registered \\
  earns' = earns \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \\
  (c? \in earns \land sinks' = sinks \oplus \{ c? \mapsto 1 \}) \lor
    (c? \notin earns \land sinks' = sinks) \\
  (c? \in earns \land c? \in lapsed \land dirty' = fset) \lor
    (\lnot (c? \in earns \land c? \in lapsed) \land dirty' = dirty) \\
  shown' = shown \\
  iphase' = iphase \\
  ifor' = ifor \\
  ibefore' = ibefore
\end{schema}

\begin{schema}{LeaseLapse}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \in registered \\
  c? \notin lapsed \\
  registered' = registered \\
  earns' = earns \\
  lapsed' = lapsed \cup \{ c? \} \\
  open' = open \\
  sinks' = sinks \\
  dirty' = dirty \\
  shown' = shown \\
  iphase' = iphase \\
  ifor' = ifor \\
  ibefore' = ibefore
\end{schema}

\begin{schema}{Departure}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \in registered \\
  registered' = registered \setminus \{ c? \} \\
  earns' = earns \setminus \{ c? \} \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \\
  sinks' = sinks \oplus \{ c? \mapsto 0 \} \\
  (c? \in earns \land dirty' = fset) \lor
    (c? \notin earns \land dirty' = dirty) \\
  shown' = shown \\
  iphase' = iphase \\
  ifor' = ifor \\
  ibefore' = ibefore
\end{schema}

\begin{zed}
  DropGraceful \defs Departure
\end{zed}

\begin{zed}
  Sweep \defs Departure \land [MenuPres; c? : CONN | c? \in lapsed]
\end{zed}

\begin{schema}{Teardown}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \in open \\
  registered' = registered \\
  earns' = earns \\
  lapsed' = lapsed \\
  open' = open \setminus \{ c? \} \\
  sinks' = sinks \\
  dirty' = dirty \\
  shown' = shown \\
  iphase' = iphase \\
  ifor' = ifor \\
  ibefore' = ibefore
\end{schema}

\begin{schema}{Push}
  \Delta MenuPres
\where
  dirty = fset \\
  registered' = registered \\
  earns' = earns \\
  lapsed' = lapsed \\
  open' = open \\
  sinks' = sinks \\
  dirty' = fclear \\
  shown' = earns \setminus lapsed \\
  iphase' = iphase \\
  ifor' = ifor \\
  ibefore' = ibefore
\end{schema}
