% FIDELITY CONTROL for docs/menu_presence.tex, defect D5.
%
% Differs from menu_presence.tex in exactly one place: Identify is split into
% IdentifyRead and IdentifyWrite around the store write lock. The shipped
% facade read `before = identity.current(scope)` outside the lock, then
% recorded the identity, then bound the sink and marked only if the identity
% changed from `before`. A departure that lands between the read and the write
% makes the re-registration look unchanged.
% (iphase/ifor/ibefore are the split-window devices, inert in the fixed spec.)
%
% Expected: BOTH negations are REACHABLE.
%   P1: probcli docs/menu_presence_d5_buggy.tex -model_check -nodead \
%         -goal "dirty = fclear & (not((earns \ lapsed) <: shown) or not(shown <: earns))" \
%         -p DEFAULT_SETSIZE 2
%   P2: probcli docs/menu_presence_d5_buggy.tex -model_check -nodead \
%         -goal "#c.(c : CONN & ((c : earns & sinks(c) /= 1) or sinks(c) > 1 \
%           or (c /: registered & sinks(c) /= 0)))" -p DEFAULT_SETSIZE 2
% Trace: Attach(c); IdentifyRead(c) [before = earning]; DropGraceful(c, {});
% Push; IdentifyWrite(c) leaves earns = {c}, sinks(c) = 0, dirty = fclear,
% shown = {}: no entry now, and a ghost when c next departs.
%
% The remaining operations are those of menu_presence.tex, whose prose
% describes them.
%

\begin{zed}
  FSTATE ::= fclear | fset
\end{zed}

\begin{zed}
  IPHASE ::= iidle | iwrite
\end{zed}

\begin{schema}{MenuPres}
  registered : \power CONN \\
  agents : \power CONN \\
  cbs : \power CONN \\
  earns : \power CONN \\
  lapsed : \power CONN \\
  open : \power CONN \\
  sinks : CONN \fun 0 \upto 2 \\
  dirty : FSTATE \\
  shown : \power CONN \\
  sdirty : \power CONN \\
  stale : \power CONN
 \\
  iphase : IPHASE \\
  ifor : \power CONN \\
  ibefore : FSTATE
\where
  agents \subseteq registered \\
  cbs \subseteq registered \\
  earns = agents \cup cbs \\
  lapsed \subseteq registered \\
  sdirty \subseteq registered \\
  stale \subseteq registered \\
  \# ifor \leq 1
\end{schema}

\begin{schema}{Init}
  MenuPres'
\where
  registered' = \emptyset \\
  agents' = \emptyset \\
  cbs' = \emptyset \\
  lapsed' = \emptyset \\
  open' = \emptyset \\
  sinks' = CONN \cross \{ 0 \} \\
  dirty' = fclear \\
  shown' = \emptyset \\
  sdirty' = \emptyset \\
  stale' = \emptyset \\
  iphase' = iidle \\
  ifor' = \emptyset \\
  ibefore' = fclear
\end{schema}

\begin{schema}{AgentRecord}
  \Delta MenuPres \\
  c? : CONN
\where
  registered' = registered \cup \{ c? \} \\
  agents' = agents \cup \{ c? \} \\
  cbs' = cbs \\
  lapsed' = lapsed \setminus \{ c? \} \\
  sinks' = sinks \oplus \{ c? \mapsto 1 \} \\
  (c? \in earns \land c? \notin lapsed \land dirty' = dirty) \lor
    ((c? \notin earns \lor c? \in lapsed) \land dirty' = fset) \\
  shown' = shown \\
  (c? \in lapsed \land sdirty' = sdirty \cup \{ c? \}) \lor
    (c? \notin lapsed \land sdirty' = sdirty) \\
  stale' = stale \\
  iphase' = iphase \\
  ifor' = ifor \\
  ibefore' = ibefore
\end{schema}

\begin{schema}{IdentifyRead}
  \Delta MenuPres \\
  c? : CONN
\where
  iphase = iidle \\
  iphase' = iwrite \\
  ifor' = \{ c? \} \\
  (c? \in earns \land ibefore' = fset) \lor
    (c? \notin earns \land ibefore' = fclear) \\
  registered' = registered \\
  agents' = agents \\
  cbs' = cbs \\
  lapsed' = lapsed \\
  open' = open \\
  sinks' = sinks \\
  dirty' = dirty \\
  shown' = shown \\
  sdirty' = sdirty \\
  stale' = stale
\end{schema}

\texttt{IdentifyRead} is the facade's \texttt{before} snapshot, taken outside the store's write lock.

\begin{schema}{IdentifyWrite}
  \Delta MenuPres \\
  c? : CONN
\where
  iphase = iwrite \\
  c? \in ifor \\
  iphase' = iidle \\
  ifor' = \emptyset \\
  ibefore' = fclear \\
  registered' = registered \cup \{ c? \} \\
  agents' = agents \cup \{ c? \} \\
  cbs' = cbs \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \\
  (ibefore = fclear \land sinks' = sinks \oplus \{ c? \mapsto 1 \}) \lor
    (ibefore = fset \land sinks' = sinks) \\
  (ibefore = fclear \land dirty' = fset) \lor
    (ibefore = fset \land dirty' = dirty) \\
  shown' = shown \\
  (c? \in lapsed \land sdirty' = sdirty \cup \{ c? \}) \lor
    (c? \notin lapsed \land sdirty' = sdirty) \\
  stale' = stale
\end{schema}

\texttt{IdentifyWrite} is the record followed by the comparison against \texttt{before}: a departure landing between the two makes the re-registration look unchanged.

  registered' = registered \setminus \{ c? \} \\
  agents' = agents \setminus \{ c? \} \\
  cbs' = cbs \setminus \{ c? \} \\

  stale' = (stale \setminus \{ c? \}) \cup r? \\
