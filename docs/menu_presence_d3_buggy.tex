% Differs from menu_presence.tex in exactly one place: ReadLoopRecord, the
% per-frame HubClientRegistry.record(conn, identity) of ws_listen._read_loop,
% is the bare registry upsert. On a session the sweep has removed (socket
% still open) it re-registers the agent -- presence returns -- but raises no
% mark and binds no sink.
%
% Expected: BOTH negations are REACHABLE.
%   P1: probcli docs/menu_presence_d3_buggy.tex -model_check -nodead \
%         -goal "dirty = fclear & (not((earns \ lapsed) <: shown) or not(shown <: earns))" \
%         -p DEFAULT_SETSIZE 2
%       Shortest trace: Attach(c); LeaseLapse(c); Push; ReadLoopRecord(c)
%       (the frame revives a lapsed, unswept session and does not mark); the
%       swept variant Attach(c); DropGraceful(c, {}); Push; ReadLoopRecord(c)
%       also leaves earns = {c}, dirty = fclear, shown = {}.
%   P2: probcli docs/menu_presence_d3_buggy.tex -model_check -nodead \
%         -goal "#c.(c : CONN & ((c : earns & sinks(c) /= 1) or sinks(c) > 1 \
%           or (c /: registered & sinks(c) /= 0)))" -p DEFAULT_SETSIZE 2
%       The revived c earns presence with sinks(c) = 0: its next departure
%       fires nothing, so the entry it was finally given is never marked away.

% describes them.

\title{lux Menu Presence, D3 control: the read loop revives a swept session}
\maketitle

\begin{zed}
  [CONN]
\end{zed}

\begin{zed}
  FSTATE ::= fclear | fset
\end{zed}

\begin{schema}{MenuPres}
  registered : \power CONN \\
  agents : \power CONN \\
  cbs : \power CONN \\
  earns : \power CONN \\
  lapsed : \power CONN \\
  open : \power CONN \\
  sinks : CONN \fun 0 \upto 2 \\
  dirty : FSTATE \\
  shown : \power CONN \\
  sdirty : \power CONN \\
  stale : \power CONN
\where
  agents \subseteq registered \\
  cbs \subseteq registered \\
  earns = agents \cup cbs \\
  lapsed \subseteq registered \\
  sdirty \subseteq registered \\
  stale \subseteq registered
\end{schema}

\begin{schema}{Init}
  MenuPres'
\where
  registered' = \emptyset \\
  agents' = \emptyset \\
  cbs' = \emptyset \\
  lapsed' = \emptyset \\
  open' = \emptyset \\
  sinks' = CONN \cross \{ 0 \} \\
  dirty' = fclear \\
  shown' = \emptyset \\
  sdirty' = \emptyset \\
  stale' = \emptyset
\end{schema}

\begin{schema}{AgentRecord}
  \Delta MenuPres \\
  c? : CONN
\where
  registered' = registered \cup \{ c? \} \\
  agents' = agents \cup \{ c? \} \\
  cbs' = cbs \\
  lapsed' = lapsed \setminus \{ c? \} \\
  sinks' = sinks \oplus \{ c? \mapsto 1 \} \\
  (c? \in earns \land c? \notin lapsed \land dirty' = dirty) \lor
    ((c? \notin earns \lor c? \in lapsed) \land dirty' = fset) \\
  shown' = shown \\
  (c? \in lapsed \land sdirty' = sdirty \cup \{ c? \}) \lor
    (c? \notin lapsed \land sdirty' = sdirty) \\
  stale' = stale
\end{schema}

\begin{schema}{BareRecord}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \in open \\
  registered' = registered \cup \{ c? \} \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \\
  sinks' = sinks \\
  dirty' = dirty \\
  agents' = agents \cup \{ c? \} \\
  cbs' = cbs \\
  shown' = shown \\
  sdirty' = sdirty \\
  stale' = stale
\end{schema}

\texttt{BareRecord} is the defect: the upsert alone, with neither mark nor sink.

\begin{zed}
  Identify \defs AgentRecord \land [\Delta MenuPres | open' = open]
\end{zed}

\begin{zed}
  Attach \defs AgentRecord \land [\Delta MenuPres; c? : CONN | open' = open \cup \{ c? \}]
\end{zed}

\begin{zed}
  ReadLoopRecord \defs BareRecord
\end{zed}

\begin{schema}{Connect}
  \Delta MenuPres \\
  c? : CONN
\where
  registered' = registered \cup \{ c? \} \\
  agents' = agents \\
  cbs' = cbs \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \\
  (c? \in earns \land sinks' = sinks \oplus \{ c? \mapsto 1 \}) \lor
    (c? \notin earns \land sinks' = sinks) \\
  (c? \in earns \land c? \in lapsed \land dirty' = fset) \lor
    (\lnot (c? \in earns \land c? \in lapsed) \land dirty' = dirty) \\
  shown' = shown \\
  (c? \in lapsed \land sdirty' = sdirty \cup \{ c? \}) \lor
    (c? \notin lapsed \land sdirty' = sdirty) \\
  stale' = stale
\end{schema}

\begin{schema}{Renew}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \in registered \\
  registered' = registered \\
  agents' = agents \\
  cbs' = cbs \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \\
  (c? \in earns \land sinks' = sinks \oplus \{ c? \mapsto 1 \}) \lor
    (c? \notin earns \land sinks' = sinks) \\
  (c? \in earns \land c? \in lapsed \land dirty' = fset) \lor
    (\lnot (c? \in earns \land c? \in lapsed) \land dirty' = dirty) \\
  shown' = shown \\
  (c? \in lapsed \land sdirty' = sdirty \cup \{ c? \}) \lor
    (c? \notin lapsed \land sdirty' = sdirty) \\
  stale' = stale
\end{schema}

\begin{schema}{LeaseLapse}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \in registered \\
  c? \notin lapsed \\
  registered' = registered \\
  agents' = agents \\
  cbs' = cbs \\
  lapsed' = lapsed \cup \{ c? \} \\
  open' = open \\
  sinks' = sinks \\
  dirty' = dirty \\
  shown' = shown \\
  sdirty' = sdirty \\
  stale' = stale
\end{schema}

\begin{schema}{AppletAttach}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \notin agents \\
  registered' = registered \cup \{ c? \} \\
  agents' = agents \\
  cbs' = cbs \setminus \{ c? \} \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \cup \{ c? \} \\
  sinks' = sinks \\
  (c? \in cbs \land dirty' = fset) \lor (c? \notin cbs \land dirty' = dirty) \\
  shown' = shown \\
  (c? \in lapsed \land sdirty' = sdirty \cup \{ c? \}) \lor
    (c? \notin lapsed \land sdirty' = sdirty) \\
  stale' = stale
\end{schema}

\begin{schema}{RegisterCallback}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \in registered \\
  c? \in open \\
  registered' = registered \\
  agents' = agents \\
  cbs' = cbs \cup \{ c? \} \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \\
  sinks' = sinks \oplus \{ c? \mapsto 1 \} \\
  (c? \in earns \land c? \notin lapsed \land dirty' = dirty) \lor
    ((c? \notin earns \lor c? \in lapsed) \land dirty' = fset) \\
  shown' = shown \\
  (c? \in lapsed \land sdirty' = sdirty \cup \{ c? \}) \lor
    (c? \notin lapsed \land sdirty' = sdirty) \\
  stale' = stale
\end{schema}

\begin{schema}{Detach}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \in open \\
  registered' = registered \\
  agents' = agents \\
  cbs' = cbs \setminus \{ c? \} \\
  lapsed' = lapsed \\
  open' = open \setminus \{ c? \} \\
  sinks' = sinks \\
  (c? \in cbs \setminus agents \land dirty' = fset) \lor
    (c? \notin cbs \setminus agents \land dirty' = dirty) \\
  shown' = shown \\
  sdirty' = sdirty \\
  stale' = stale
\end{schema}

\begin{schema}{Departure}
  \Delta MenuPres \\
  c? : CONN \\
  r? : \power CONN
\where
  c? \in registered \\
  r? \subseteq registered \setminus \{ c? \} \\
  registered' = registered \setminus \{ c? \} \\
  agents' = agents \setminus \{ c? \} \\
  cbs' = cbs \setminus \{ c? \} \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \\
  sinks' = sinks \oplus \{ c? \mapsto 0 \} \\
  (c? \in earns \land sinks~c? \geq 1 \land dirty' = fset) \lor
    (\lnot (c? \in earns \land sinks~c? \geq 1) \land dirty' = dirty) \\
  shown' = shown \\
  sdirty' = (sdirty \setminus \{ c? \}) \cup r? \\
  stale' = (stale \setminus \{ c? \}) \cup r?
\end{schema}

\begin{zed}
  DropGraceful \defs Departure
\end{zed}

\begin{zed}
  Sweep \defs Departure \land [MenuPres; c? : CONN | c? \in lapsed]
\end{zed}

\begin{schema}{Push}

  cbs' = cbs \\
  lapsed' = lapsed \\
  open' = open \\
  sinks' = sinks \\

  sdirty' = sdirty \\
  stale' = stale
\end{schema}

  cbs' = cbs \\

\begin{schema}{PushScenes}
  \Delta MenuPres
\where
  sdirty \neq \emptyset \\
  registered' = registered \\
  agents' = agents \\
  cbs' = cbs \\
  lapsed' = lapsed \\
  open' = open \\
  sinks' = sinks \\
  dirty' = dirty \\
  shown' = shown \\
  sdirty' = \emptyset \\
  stale' = (stale \setminus (sdirty \setminus lapsed)) \cup (sdirty \cap lapsed)
\end{schema}
