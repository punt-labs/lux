% Differs from menu_presence.tex in exactly one place: Attach binds a NEW sink
% each time (ws_listen binds `lambda _: self._menus.mark_menus()`, a fresh
% object per attach, and DepartureSinks.bind dedups by equality, which two
% lambdas never satisfy). The count saturates at two, where the leak is
% already established; the guard keeps the state space finite.
%
% Expected: P2's negation is REACHABLE (the "no more than one sink" conjunct).
%   probcli docs/menu_presence_d6_buggy.tex -model_check -nodead \
%     -goal "#c.(c : CONN & ((c : earns & sinks(c) /= 1) or sinks(c) > 1 \
%       or (c /: registered & sinks(c) /= 0)))" -p DEFAULT_SETSIZE 3
% Trace: Attach(c); Attach(c) -- a reconnect of the same identity -- leaves
% sinks(c) = 2. P1 is untouched: the leak is silent, which is why no
% behavioural test of the menu finds it.
%
% The remaining operations are those of menu_presence.tex, whose prose
% describes them.
%

\documentclass[a4paper,10pt,fleqn]{article}
\usepackage[margin=1in]{geometry}
\usepackage{fuzz}

\begin{document}

\title{lux Menu Presence, D6 control: a sink bound per attach}
\maketitle

\begin{zed}
  [CONN]
\end{zed}

\begin{zed}
  FSTATE ::= fclear | fset
\end{zed}

\begin{schema}{MenuPres}
  registered : \power CONN \\
  earns : \power CONN \\
  lapsed : \power CONN \\
  open : \power CONN \\
  sinks : CONN \fun 0 \upto 2 \\
  dirty : FSTATE \\
  shown : \power CONN
\where
  earns \subseteq registered \\
  lapsed \subseteq registered
\end{schema}

\begin{schema}{Init}
  MenuPres'
\where
  registered' = \emptyset \\
  earns' = \emptyset \\
  lapsed' = \emptyset \\
  open' = \emptyset \\
  sinks' = CONN \cross \{ 0 \} \\
  dirty' = fclear \\
  shown' = \emptyset
\end{schema}

\begin{schema}{AgentRecord}
  \Delta MenuPres \\
  c? : CONN
\where
  registered' = registered \cup \{ c? \} \\
  earns' = earns \cup \{ c? \} \\
  lapsed' = lapsed \setminus \{ c? \} \\
  sinks' = sinks \oplus \{ c? \mapsto 1 \} \\
  (c? \in earns \land dirty' = dirty) \lor
    (c? \notin earns \land dirty' = fset) \\
  shown' = shown
\end{schema}

\begin{schema}{LeakyRecord}
  \Delta MenuPres \\
  c? : CONN
\where
  sinks(c?) < 2 \\
  registered' = registered \cup \{ c? \} \\
  earns' = earns \cup \{ c? \} \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \cup \{ c? \} \\
  sinks' = sinks \oplus \{ c? \mapsto sinks(c?) + 1 \} \\
  (c? \in earns \land dirty' = dirty) \lor
    (c? \notin earns \land dirty' = fset) \\
  shown' = shown
\end{schema}

\texttt{LeakyRecord} is the defect: the bind adds a sink instead of being
idempotent.

\begin{zed}
  Identify \defs AgentRecord \land [\Delta MenuPres | open' = open]
\end{zed}

\begin{zed}
  Attach \defs LeakyRecord
\end{zed}

\begin{zed}
  ReadLoopRecord \defs AgentRecord \land
    [\Delta MenuPres; c? : CONN | open' = open \land c? \in open]
\end{zed}

\begin{schema}{Connect}
  \Delta MenuPres \\
  c? : CONN
\where
  registered' = registered \cup \{ c? \} \\
  earns' = earns \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \\
  sinks' = sinks \\
  dirty' = dirty \\
  shown' = shown
\end{schema}

\begin{schema}{Renew}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \in registered \\
  registered' = registered \\
  earns' = earns \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \\
  sinks' = sinks \\
  dirty' = dirty \\
  shown' = shown
\end{schema}

\begin{schema}{LeaseLapse}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \in registered \\
  c? \notin lapsed \\
  registered' = registered \\
  earns' = earns \\
  lapsed' = lapsed \cup \{ c? \} \\
  open' = open \\
  sinks' = sinks \\
  dirty' = dirty \\
  shown' = shown
\end{schema}

\begin{schema}{Departure}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \in registered \\
  registered' = registered \setminus \{ c? \} \\
  earns' = earns \setminus \{ c? \} \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \\
  sinks' = sinks \oplus \{ c? \mapsto 0 \} \\
  (c? \in earns \land dirty' = fset) \lor
    (c? \notin earns \land dirty' = dirty) \\
  shown' = shown
\end{schema}

\begin{zed}
  DropGraceful \defs Departure
\end{zed}

\begin{zed}
  Sweep \defs Departure \land [MenuPres; c? : CONN | c? \in lapsed]
\end{zed}

\begin{schema}{Teardown}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \in open \\
  registered' = registered \\
  earns' = earns \\
  lapsed' = lapsed \\
  open' = open \setminus \{ c? \} \\
  sinks' = sinks \\
  dirty' = dirty \\
  shown' = shown
\end{schema}

\begin{schema}{Push}
  \Delta MenuPres
\where
  dirty = fset \\
  registered' = registered \\
  earns' = earns \\
  lapsed' = lapsed \\
  open' = open \\
  sinks' = sinks \\
  dirty' = fclear \\
  shown' = earns
\end{schema}
