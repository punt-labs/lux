\section{Constants and Free Types}
% ============================================================

Three generations: the original, the replacement, and one more, so that a
replacement can itself be replaced while an abandoned reader of the first is
still alive.  Two requests: the least that lets a late reply, or the ack of an
un-awaited scene, meet a later request in one generation.  One buffered reply
is the least that lets an abandoned reader have something left to route.  The
sets of readers \texttt{T} and of callers \texttt{C} are given sets, sized on
the command line (Section~\ref{sec:verify}).

\begin{axdef}
  maxGen : \nat \\
  maxReq : \nat \\
  maxBuf : \nat
\where
  maxGen = 3 \\
  maxReq = 2 \\
  maxBuf = 1
\end{axdef}

\begin{zed}
  [T]
\end{zed}

\begin{zed}
  [C]
\end{zed}

\begin{zed}
  K ::= kp | ks | kq
\end{zed}

\begin{zed}
  LST ::= lexit | lrun | lstop
\end{zed}

\begin{zed}
  FSTATE ::= fclear | fset
\end{zed}

\begin{zed}
  CST ::= cidle | clockwait | csent | cwait
\end{zed}

\section{State}
% ============================================================

\begin{schema}{Conn}
  curGen : \nat \\
  held : FSTATE \\
  started : \power \nat
\where
  started \subseteq 1 \upto maxGen \\
  curGen \leq maxGen \\
  held = fset \implies curGen > 0
\end{schema}

\begin{schema}{ReqT}
  nreq : \nat \\
  rk : (0 \upto maxReq) \fun K \\
  rgen : (0 \upto maxReq) \fun (0 \upto maxGen)
\where
  nreq \leq maxReq
\end{schema}

\begin{schema}{Cal}
  cstate : C \fun CST \\
  ck : C \fun K \\
  creq : C \fun \nat
\where
  \forall c : C \spot creq~c \leq maxReq
\end{schema}

\begin{schema}{Snk}
  owed : \power \nat \\
  led : \power \nat \\
  mbox : C \pfun \nat
\where
  owed \subseteq 1 \upto maxReq \\
  led \subseteq 1 \upto maxReq \\
  \ran mbox \subseteq 1 \upto maxReq
\end{schema}

\begin{schema}{Thr}
  lst : T \fun LST \\
  lgen : T \fun \nat \\
  armed : T \fun FSTATE \\
  bufr : T \fun (\power \nat) \\
  afterStop : T \fun \nat
\where
  \forall t : T \spot lgen~t \leq maxGen \land \# (bufr~t) \leq maxBuf
    \land bufr~t \subseteq 1 \upto maxReq \land afterStop~t \leq maxBuf + 1
\end{schema}

\begin{schema}{Hist}
  broken : \power \nat \\
  deliberate : \power \nat \\
  crossGen : FSTATE \\
  crossReq : FSTATE \\
  wrongRetire : FSTATE \\
  starved : FSTATE
\where
  broken \subseteq 0 \upto maxGen \\
  deliberate \subseteq 0 \upto maxGen
\end{schema}

\begin{schema}{LL}
  Conn \\
  ReqT \\
  Cal \\
  Snk \\
  Thr \\
  Hist
\end{schema}
