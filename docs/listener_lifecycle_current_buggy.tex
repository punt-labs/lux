\section{Constants and Free Types}

\begin{axdef}
  maxGen : \nat \\
  maxReq : \nat \\
  maxBuf : \nat
\where
  maxGen = 3 \\
  maxReq = 2 \\
  maxBuf = 1
\end{axdef}

\begin{zed}
  T ::= ta | tb
\end{zed}

\begin{zed}
  LST ::= lexit | lrun | lstop
\end{zed}

\begin{zed}
  FSTATE ::= fclear | fset
\end{zed}

\begin{zed}
  CST ::= cidle | csent | cwait
\end{zed}

\section{State}

\begin{schema}{Conn}
  curGen : \nat \\
  held : FSTATE
\where
  curGen \leq maxGen \\
  held = fset \implies curGen > 0
\end{schema}

\begin{schema}{Req}
  nreq : \nat \\
  creq : \nat \\
  cstate : CST \\
  rgen : (0 \upto maxReq) \fun (0 \upto maxGen) \\
  owed : \power \nat \\
  qrep : \power \nat
\where
  nreq \leq maxReq \\
  creq \leq maxReq \\
  owed \subseteq 1 \upto maxReq \\
  qrep \subseteq 1 \upto maxReq
\end{schema}

\begin{schema}{Thr}
  lst : T \fun LST \\
  lgen : T \fun \nat \\
  armed : T \fun FSTATE \\
  bufr : T \fun (\power \nat) \\
  afterStop : T \fun \nat
\where
  \forall t : T \spot lgen~t \leq maxGen \land \# (bufr~t) \leq maxBuf
    \land bufr~t \subseteq 1 \upto maxReq \land afterStop~t \leq maxBuf + 1
\end{schema}

\begin{schema}{Hist}
  broken : \power \nat \\
  deliberate : \power \nat \\
  crossGen : FSTATE \\
  crossReq : FSTATE
\where
  broken \subseteq 0 \upto maxGen \\
  deliberate \subseteq 0 \upto maxGen
\end{schema}

\begin{schema}{LL}
  Conn \\
  Req \\
  Thr \\
  Hist
\end{schema}

\begin{schema}{Init}
  LL'
\where
  curGen' = 0 \\
  held' = fclear \\
  nreq' = 0 \\
  creq' = 0 \\
  cstate' = cidle \\
  \forall i : 0 \upto maxReq \spot rgen'~i = 0 \\
  owed' = \emptyset \\
  qrep' = \emptyset \\
  lst' = \{ ta \mapsto lexit, tb \mapsto lexit \} \\
  lgen' = \{ ta \mapsto 0, tb \mapsto 0 \} \\
  armed' = \{ ta \mapsto fclear, tb \mapsto fclear \} \\
  bufr' = \{ ta \mapsto \emptyset, tb \mapsto \emptyset \} \\
  afterStop' = \{ ta \mapsto 0, tb \mapsto 0 \} \\
  broken' = \emptyset \\
  deliberate' = \emptyset \\
  crossGen' = fclear \\
  crossReq' = fclear
\end{schema}
