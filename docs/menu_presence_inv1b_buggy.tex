% menu_presence_inv1b_buggy.tex
%
% FIDELITY CONTROL for docs/menu_presence.tex, the trap in INV-1's fix.
%
% Differs from menu_presence.tex in exactly one place: Detach and AppletAttach
% (the two steps that withdraw callbacks) raise the menu mark only when the
% session was LIVE, as ClientLifecycle's announce does when it is driven by
% MenuListing.change_to ("holds" reads the lease). A lapsed applet whose entry
% is still on the Display, and whose callbacks are then withdrawn, is neither
% listed before nor after, so nothing marks; the sweep that later departs the
% now non-earning session fires no sink worth the name, and the entry stays.
%
% Expected: the P1 negation is REACHABLE.
%   probcli docs/menu_presence_inv1b_buggy.tex -model_check -nodead \
%     -goal "dirty = fclear & (not((earns \ lapsed) <: shown) or not(shown <: earns))" \
%     -p DEFAULT_SETSIZE 2
% Trace: AppletAttach(c); RegisterCallback(c); Push; LeaseLapse(c); AppletAttach(c)
% leaves earns = {}, shown = {c}, c registered and LIVE, dirty = fclear: the
% session keeps renewing, is never swept, and the entry stays for good. The
% stronger goal reaches it directly:
%   -goal "dirty = fclear & #c.(c : shown & c : registered & c /: lapsed & c /: earns)"
% (Detach(c) in place of the last step leaves the lapsed variant.)
%
% The remaining operations are those of menu_presence.tex, whose prose
% describes them.
%

\documentclass[a4paper,10pt,fleqn]{article}
\usepackage[margin=1in]{geometry}
\usepackage{fuzz}

\begin{document}

\title{lux Menu Presence, INV-1b control: a lease-gated withdrawal mark}
\maketitle

\begin{zed}
  [CONN]
\end{zed}

\begin{zed}
  FSTATE ::= fclear | fset
\end{zed}

\begin{schema}{MenuPres}
  registered : \power CONN \\
  agents : \power CONN \\
  cbs : \power CONN \\
  earns : \power CONN \\
  lapsed : \power CONN \\
  open : \power CONN \\
  sinks : CONN \fun 0 \upto 2 \\
  dirty : FSTATE \\
  shown : \power CONN \\
  sdirty : \power CONN \\
  stale : \power CONN
\where
  agents \subseteq registered \\
  cbs \subseteq registered \\
  earns = agents \cup cbs \\
  lapsed \subseteq registered \\
  sdirty \subseteq registered \\
  stale \subseteq registered
\end{schema}

\begin{schema}{Init}
  MenuPres'
\where
  registered' = \emptyset \\
  agents' = \emptyset \\
  cbs' = \emptyset \\
  lapsed' = \emptyset \\
  open' = \emptyset \\
  sinks' = CONN \cross \{ 0 \} \\
  dirty' = fclear \\
  shown' = \emptyset \\
  sdirty' = \emptyset \\
  stale' = \emptyset
\end{schema}

\begin{schema}{AgentRecord}
  \Delta MenuPres \\
  c? : CONN
\where
  registered' = registered \cup \{ c? \} \\
  agents' = agents \cup \{ c? \} \\
  cbs' = cbs \\
  lapsed' = lapsed \setminus \{ c? \} \\
  sinks' = sinks \oplus \{ c? \mapsto 1 \} \\
  (c? \in earns \land c? \notin lapsed \land dirty' = dirty) \lor
    ((c? \notin earns \lor c? \in lapsed) \land dirty' = fset) \\
  shown' = shown \\
  (c? \in lapsed \land sdirty' = sdirty \cup \{ c? \}) \lor
    (c? \notin lapsed \land sdirty' = sdirty) \\
  stale' = stale
\end{schema}

\begin{zed}
  Identify \defs AgentRecord \land [\Delta MenuPres | open' = open]
\end{zed}

\begin{zed}
  Attach \defs AgentRecord \land [\Delta MenuPres; c? : CONN | open' = open \cup \{ c? \}]
\end{zed}

\begin{zed}
  ReadLoopRecord \defs AgentRecord \land
    [\Delta MenuPres; c? : CONN | open' = open \land c? \in open]
\end{zed}

\begin{schema}{Connect}
  \Delta MenuPres \\
  c? : CONN
\where
  registered' = registered \cup \{ c? \} \\
  agents' = agents \\
  cbs' = cbs \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \\
  (c? \in earns \land sinks' = sinks \oplus \{ c? \mapsto 1 \}) \lor
    (c? \notin earns \land sinks' = sinks) \\
  (c? \in earns \land c? \in lapsed \land dirty' = fset) \lor
    (\lnot (c? \in earns \land c? \in lapsed) \land dirty' = dirty) \\
  shown' = shown \\
  (c? \in lapsed \land sdirty' = sdirty \cup \{ c? \}) \lor
    (c? \notin lapsed \land sdirty' = sdirty) \\
  stale' = stale
\end{schema}

\begin{schema}{Renew}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \in registered \\
  registered' = registered \\
  agents' = agents \\
  cbs' = cbs \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \\
  (c? \in earns \land sinks' = sinks \oplus \{ c? \mapsto 1 \}) \lor
    (c? \notin earns \land sinks' = sinks) \\
  (c? \in earns \land c? \in lapsed \land dirty' = fset) \lor
    (\lnot (c? \in earns \land c? \in lapsed) \land dirty' = dirty) \\
  shown' = shown \\
  (c? \in lapsed \land sdirty' = sdirty \cup \{ c? \}) \lor
    (c? \notin lapsed \land sdirty' = sdirty) \\
  stale' = stale
\end{schema}

\begin{schema}{LeaseLapse}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \in registered \\
  c? \notin lapsed \\
  registered' = registered \\
  agents' = agents \\
  cbs' = cbs \\
  lapsed' = lapsed \cup \{ c? \} \\
  open' = open \\
  sinks' = sinks \\
  dirty' = dirty \\
  shown' = shown \\
  sdirty' = sdirty \\
  stale' = stale
\end{schema}

\begin{schema}{AppletAttach}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \notin agents \\
  registered' = registered \cup \{ c? \} \\
  agents' = agents \\
  cbs' = cbs \setminus \{ c? \} \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \cup \{ c? \} \\
  sinks' = sinks \\
  (c? \in cbs \land c? \notin lapsed \land dirty' = fset) \lor
    (\lnot (c? \in cbs \land c? \notin lapsed) \land dirty' = dirty) \\
  shown' = shown \\
  (c? \in lapsed \land sdirty' = sdirty \cup \{ c? \}) \lor
    (c? \notin lapsed \land sdirty' = sdirty) \\
  stale' = stale
\end{schema}

\begin{schema}{RegisterCallback}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \in registered \\
  c? \in open \\
  registered' = registered \\
  agents' = agents \\
  cbs' = cbs \cup \{ c? \} \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \\
  sinks' = sinks \oplus \{ c? \mapsto 1 \} \\
  (c? \in earns \land c? \notin lapsed \land dirty' = dirty) \lor
    ((c? \notin earns \lor c? \in lapsed) \land dirty' = fset) \\
  shown' = shown \\
  (c? \in lapsed \land sdirty' = sdirty \cup \{ c? \}) \lor
    (c? \notin lapsed \land sdirty' = sdirty) \\
  stale' = stale
\end{schema}

\begin{schema}{Detach}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \in open \\
  registered' = registered \\
  agents' = agents \\
  cbs' = cbs \setminus \{ c? \} \\
  lapsed' = lapsed \\
  open' = open \setminus \{ c? \} \\
  sinks' = sinks \\
  (c? \in cbs \setminus agents \land c? \notin lapsed \land dirty' = fset) \lor
    (\lnot (c? \in cbs \setminus agents \land c? \notin lapsed) \land dirty' = dirty) \\
  shown' = shown \\
  sdirty' = sdirty \\
  stale' = stale
\end{schema}

\begin{schema}{Redeclare}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \in agents \\
  registered' = registered \\
  agents' = agents \setminus \{ c? \} \\
  cbs' = cbs \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \\
  sinks' = sinks \\
  ((c? \in lapsed \lor c? \notin cbs) \land dirty' = fset) \lor
    (\lnot (c? \in lapsed \lor c? \notin cbs) \land dirty' = dirty) \\
  shown' = shown \\
  (c? \in lapsed \land sdirty' = sdirty \cup \{ c? \}) \lor
    (c? \notin lapsed \land sdirty' = sdirty) \\
  stale' = stale
\end{schema}

\begin{schema}{Departure}
  \Delta MenuPres \\
  c? : CONN \\
  r? : \power CONN
\where
  c? \in registered \\
  r? \subseteq registered \setminus \{ c? \} \\
  registered' = registered \setminus \{ c? \} \\
  agents' = agents \setminus \{ c? \} \\
  cbs' = cbs \setminus \{ c? \} \\
  lapsed' = lapsed \setminus \{ c? \} \\
  open' = open \\
  sinks' = sinks \oplus \{ c? \mapsto 0 \} \\
  (sinks~c? \geq 1 \land dirty' = fset) \lor
    (sinks~c? < 1 \land dirty' = dirty) \\
  shown' = shown \\
  sdirty' = (sdirty \setminus \{ c? \}) \cup r? \\
  stale' = (stale \setminus \{ c? \}) \cup r?
\end{schema}

\begin{zed}
  DropGraceful \defs Departure
\end{zed}

\begin{zed}
  Sweep \defs Departure \land [MenuPres; c? : CONN | c? \in lapsed]
\end{zed}

\begin{schema}{Push}
  \Delta MenuPres
\where
  dirty = fset \\
  registered' = registered \\
  agents' = agents \\
  cbs' = cbs \\
  lapsed' = lapsed \\
  open' = open \\
  sinks' = sinks \\
  dirty' = fclear \\
  shown' = earns \setminus lapsed \\
  sdirty' = sdirty \\
  stale' = stale
\end{schema}

\begin{schema}{ScenesTouched}
  \Delta MenuPres \\
  c? : CONN
\where
  c? \in registered \\
  registered' = registered \\
  agents' = agents \\
  cbs' = cbs \\
  lapsed' = lapsed \\
  open' = open \\
  sinks' = sinks \\
  dirty' = dirty \\
  shown' = shown \\
  sdirty' = sdirty \cup \{ c? \} \\
  stale' = stale
\end{schema}

\begin{schema}{PushScenes}
  \Delta MenuPres
\where
  sdirty \neq \emptyset \\
  registered' = registered \\
  agents' = agents \\
  cbs' = cbs \\
  lapsed' = lapsed \\
  open' = open \\
  sinks' = sinks \\
  dirty' = dirty \\
  shown' = shown \\
  sdirty' = \emptyset \\
  stale' = (stale \setminus (sdirty \setminus lapsed)) \cup (sdirty \cap lapsed)
\end{schema}
