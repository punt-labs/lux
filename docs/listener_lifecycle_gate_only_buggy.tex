% FIDELITY CONTROL for docs/listener_lifecycle.tex: the stopping flag as the
% only repair.  This is the fix the round-3 reviewers proposed (R3-1, SF3-2):
% pass the thread's own `stopping` event into _routed and _read_ended, and do
% not route or report once it is set.  Here that is the permission discipline
% of the fixed model -- a reader acts only with a permission issued while its
% flag was clear -- and a reader holds its own generation's socket (Feed), but
% the reply sink is still one queue shared across reconnects, a caller takes
% any reply, a report is still `end`, and the fixed model's fourth
% StartListener guard (retire the socket before a successor) is NOT present.
%
% Differences from listener_lifecycle.tex: RouteShared for RouteLive and
% RouteDead; BrokenWrong and BrokenQuiet for BrokenStale; AwaitShared,
% AwaitFailClosed and ConsumeAny for the per-generation wait; TimeoutPlain;
% no fourth guard on StartListener.
%
% Expected: L1, L2, L5, L6 VIOLATED (the flag cannot close the window between
% the look and the act); L4a VIOLATED (the flag does nothing for it: it is
% the missing guard); L3 and L4b hold (the permission discipline and the
% own-socket read are all that they need).  Close needs no reader in lrun, as
% in the production control: a close follows a stop, so the permit race is the
% only route to the replacement.
%
% Goals (negations of the invariants; see listener_lifecycle.tex):
%   L1 (qrep /= {} & (held = fclear or {r | r:qrep & rgen(r) /= curGen} /= {}))
%        or (cstate = cwait & (held = fclear or rgen(creq) /= curGen))
%   L2 held = fclear & curGen > 0 & curGen /: (broken \/ deliberate)
%   L3 afterStop(ta) > 1 or afterStop(tb) > 1
%   L4a lst(ta) /= lexit & lst(tb) /= lexit & lgen(ta) = lgen(tb)
%   L4b {r | r:bufr(ta) & rgen(r) /= lgen(ta)} /= {}
%        or {r | r:bufr(tb) & rgen(r) /= lgen(tb)} /= {}
%   L5 crossGen = fset      L6 crossReq = fset
% Check:  fuzz FILE ; probcli FILE -model_check -nodead -p DEFAULT_SETSIZE 2 \
%           -p MAX_OPERATIONS 200 -goal "<negation>"
%

\section{Constants and Free Types}

\begin{axdef}
  maxGen : \nat \\
  maxReq : \nat \\
  maxBuf : \nat
\where
  maxGen = 3 \\
  maxReq = 2 \\
  maxBuf = 1
\end{axdef}

\begin{zed}
  T ::= ta | tb
\end{zed}

\begin{zed}
  LST ::= lexit | lrun | lstop
\end{zed}

\begin{zed}
  FSTATE ::= fclear | fset
\end{zed}

\begin{zed}
  CST ::= cidle | csent | cwait
\end{zed}

\section{State}

\begin{schema}{Conn}
  curGen : \nat \\
  held : FSTATE
\where
  curGen \leq maxGen \\
  held = fset \implies curGen > 0
\end{schema}

\begin{schema}{Req}
  nreq : \nat \\
  creq : \nat \\
  cstate : CST \\
  rgen : (0 \upto maxReq) \fun (0 \upto maxGen) \\
  owed : \power \nat \\
  qrep : \power \nat
\where
  nreq \leq maxReq \\
  creq \leq maxReq \\
  owed \subseteq 1 \upto maxReq \\
  qrep \subseteq 1 \upto maxReq
\end{schema}

\begin{schema}{Thr}
  lst : T \fun LST \\
  lgen : T \fun \nat \\
  armed : T \fun FSTATE \\
  bufr : T \fun (\power \nat) \\
  afterStop : T \fun \nat
\where
  \forall t : T \spot lgen~t \leq maxGen \land \# (bufr~t) \leq maxBuf
    \land bufr~t \subseteq 1 \upto maxReq \land afterStop~t \leq maxBuf + 1
\end{schema}

\begin{schema}{Hist}
  broken : \power \nat \\
  deliberate : \power \nat \\
  crossGen : FSTATE \\
  crossReq : FSTATE
\where
  broken \subseteq 0 \upto maxGen \\
  deliberate \subseteq 0 \upto maxGen
\end{schema}

\begin{schema}{LL}
  Conn \\
  Req \\
  Thr \\
  Hist
\end{schema}

\begin{schema}{Init}
  LL'
\where
  curGen' = 0 \\
  held' = fclear \\
  nreq' = 0 \\
  creq' = 0 \\
  cstate' = cidle \\
  \forall i : 0 \upto maxReq \spot rgen'~i = 0 \\
  owed' = \emptyset \\
  qrep' = \emptyset \\
  lst' = \{ ta \mapsto lexit, tb \mapsto lexit \} \\
  lgen' = \{ ta \mapsto 0, tb \mapsto 0 \} \\
  armed' = \{ ta \mapsto fclear, tb \mapsto fclear \} \\
  bufr' = \{ ta \mapsto \emptyset, tb \mapsto \emptyset \} \\
  afterStop' = \{ ta \mapsto 0, tb \mapsto 0 \} \\
  broken' = \emptyset \\
  deliberate' = \emptyset \\
  crossGen' = fclear \\
  crossReq' = fclear
\end{schema}

\section{Operations}

\begin{schema}{Dial}
  \Delta LL \\
  \Xi Req \\
  \Xi Thr \\
  \Xi Hist
\where
  held = fclear \\
  curGen < maxGen \\
  curGen' = curGen + 1 \\
  held' = fset
\end{schema}

\begin{schema}{Close}
  \Delta LL \\
  \Xi Thr
\where
  held = fset \\
  \forall u : T \spot lst~u \neq lrun \\
  held' = fclear \\
  curGen' = curGen \\
  qrep' = \emptyset \\
  owed' = \{ r : owed | rgen~r \neq curGen \} \\
  nreq' = nreq \\
  creq' = creq \\
  rgen' = rgen \\
  (cstate = cwait \implies cstate' = cidle) \\
  (cstate \neq cwait \implies cstate' = cstate) \\
  deliberate' = deliberate \cup \{ curGen \} \\
  broken' = broken \\
  crossGen' = crossGen \\
  crossReq' = crossReq
\end{schema}
